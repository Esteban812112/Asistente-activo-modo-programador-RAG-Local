# Documentación de funciones
def calcular_area_circulo(radio):
    """
    Calcula el área de un círculo.
    
    Args:
        radio (float): Radio del círculo
        
    Returns:
        float: Área del círculo
        
    Ejemplo:
        >>> calcular_area_circulo(5)
        78.53981633974483
    """
    import math
    return math.pi * radio ** 2
